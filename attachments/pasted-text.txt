        function updateSimulation() {
            const T0 = parseFloat(document.getElementById('capital-slider').value);
            const k = parseFloat(document.getElementById('rate-slider').value);
            const timeVal = parseInt(document.getElementById('years-slider').value);

            document.getElementById('capital-val').innerText = `${T0}`;
            document.getElementById('rate-val').innerText = `${k.toFixed(2)}`;
            document.getElementById('years-val').innerText = `${timeVal}`;

            const labels = [];
            const data = [];
            const stepTime = Math.max(1, Math.floor(timeVal / 5));

            for (let i = 0; i <= timeVal; i += stepTime) {
                labels.push(`${i}m`);
                const val = 100 - (100 - T0) * Math.exp(-k * i);
                data.push(parseFloat(val.toFixed(2)));
            }

            const finalVal = 100 - (100 - T0) * Math.exp(-k * timeVal);
            const rateVal = (100 - T0) * k * Math.exp(-k * timeVal);

            document.getElementById('final-result').innerText = `${finalVal.toFixed(2)} units`;
            document.getElementById('rate-result').innerText = `${rateVal.toFixed(2)} units/min`;

            sandboxChart.data.labels = labels;
            sandboxChart.data.datasets[0].data = data;
            sandboxChart.update();
        }

        document.getElementById('capital-slider').addEventListener('input', updateSimulation);
        document.getElementById('rate-slider').addEventListener('input', updateSimulation);
        document.getElementById('years-slider').addEventListener('input', updateSimulation);

                if (category === 'all' || card.getAttribute('data-category') === category) {
                    card.style.display = 'grid';
                } else {
                    card.style.display = 'none';
                }
            });
        }
